//model
class shirt(models.Model):
    text_color = models.CharField(max_length=20)
    text_size = models.IntegerField()
    position = models.JSONField()
    rotation = models.JSONField()
    custom_text = models.TextField()
    font_family = models.CharField(max_length=100)
    shirt_color = models.CharField(max_length=20)
    image = models.URLField(blank=True, null=True)
    image_scale = models.JSONField()
    image_position = models.JSONField()
    image_rotation = models.JSONField()
    created_at = models.DateTimeField(auto_now_add=True)
    def __str__(self):
        return f"{self.tshirt_color} shirt with text '{self.custom_text}'"
    
//Serializers
class shirtSerializer(serializers.ModelSerializer):
    class Meta:
        model = shirt
        fields = '__all__'

//view
class shirtViewSet(viewsets.ModelViewSet):
    queryset = shirt.objects.all()
    serializer_class = shirtSerializer
